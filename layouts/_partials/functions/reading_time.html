{{- /* Exclude appendix and reference sections, resuming at the next peer or higher heading. Work with rendered headings so code examples and inline mentions still count. */ -}} {{- $content := "" -}} {{- $skipLevel := 0 -}} {{- $hasSkipped := false -}} {{- $sections := split (replaceRE `(?i)(])` "\x00${1}" .Content) "\x00" -}} {{- range $sections -}} {{- $heading := "" -}} {{- range findRE `(?is)^]*>.*?` . 1 -}} {{- $heading = . -}} {{- end -}} {{- with $heading -}} {{- $level := int (substr . 2 1) -}} {{- if and (gt $skipLevel 0) (le $level $skipLevel) -}} {{- $skipLevel = 0 -}} {{- end -}} {{- $title := plainify . | htmlUnescape | strings.TrimSpace -}} {{- if and (eq $skipLevel 0) (or (findRE `(?i)^(appendi(?:x|ces)|references)(?:\s|:|$)` $title) (findRE `(?i)\bid=["'](appendices|references)["']` .)) -}} {{- $skipLevel = $level -}} {{- $hasSkipped = true -}} {{- end -}} {{- end -}} {{- if eq $skipLevel 0 -}} {{- $content = printf "%s%s" $content . -}} {{- end -}} {{- end -}} {{- $minutes := .ReadingTime -}} {{- if $hasSkipped -}} {{- /* Match Hugo's default rate of 212 words per minute, rounded up. */ -}} {{- $minutes = int (math.Ceil (div (float (countwords (plainify $content | htmlUnescape))) 212.0)) -}} {{- end -}} {{- return (math.Max 1 $minutes | int) -}}
